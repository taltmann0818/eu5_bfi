bfl_pay_deposit_01 = {
if = { limit = { var:bfl_dep_01_active = 1 var:bfl_dep_01_withdrawal_requested > 0 exists = var:bfl_dep_01_party }
    save_scope_as = bfl_bank
    var:bfl_dep_01_party = { save_scope_as = bfl_party }
    set_variable = { name = bfl_payment value = { value = gold subtract = 25 subtract = var:bfl_cash_committed min = 0 max = var:bfl_dep_01_withdrawal_requested } }
    add_gold = bfl_negative_payment
    scope:bfl_party = { add_gold = scope:bfl_bank.var:bfl_payment }
    set_variable = { name = bfl_interest_paid value = { value = var:bfl_payment max = var:bfl_dep_01_interest_due } }
    change_variable = { name = bfl_dep_01_interest_due subtract = var:bfl_interest_paid }
    change_variable = { name = bfl_dep_01_withdrawal_requested subtract = var:bfl_payment }
    change_variable = { name = bfl_payment subtract = var:bfl_interest_paid }
    change_variable = { name = bfl_dep_01_principal subtract = var:bfl_payment }
    if = { limit = { var:bfl_dep_01_principal <= 0 var:bfl_dep_01_interest_due <= 0 }
        set_variable = { name = bfl_dep_01_active value = 0 } remove_variable = bfl_dep_01_party
    }
}
}
