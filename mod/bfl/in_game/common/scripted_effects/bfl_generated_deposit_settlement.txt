bfl_pay_deposit_01 = {
save_scope_as = bfl_bank if = { limit = { exists = var:bfl_dep_01_party country_exists = var:bfl_dep_01_party }
var:bfl_dep_01_party = { set_variable = { name = bfl_pool value = scope:bfl_bank.var:bfl_dep_01_pool } } }
if = { limit = { var:bfl_dep_01_active = 1 var:bfl_dep_01_withdrawal_requested > 0 exists = var:bfl_dep_01_party country_exists = var:bfl_dep_01_party
var:bfl_dep_01_party = { bfl_valid_deposit_pool = yes } }
    set_variable = { name = bfl_payment value = { value = gold subtract = 25 subtract = var:bfl_cash_committed
        min = 0 max = var:bfl_dep_01_withdrawal_requested multiply = 100 floor = yes divide = 100 } }
    add_gold = bfl_negative_payment
    var:bfl_dep_01_party = { bfl_credit_depositor = yes }
    set_variable = { name = bfl_interest_paid value = { value = var:bfl_payment max = var:bfl_dep_01_interest_due } }
    if = { limit = { var:bfl_dep_01_pool = 0 } var:bfl_dep_01_party = { bfl_record_withdrawn_interest = yes } }
    change_variable = { name = bfl_dep_01_interest_due subtract = var:bfl_interest_paid }
    change_variable = { name = bfl_dep_01_withdrawal_requested subtract = var:bfl_payment }
    change_variable = { name = bfl_payment subtract = var:bfl_interest_paid }
    change_variable = { name = bfl_dep_01_principal subtract = var:bfl_payment }
    if = { limit = { var:bfl_dep_01_principal <= 0 var:bfl_dep_01_interest_due <= 0 }
        set_variable = { name = bfl_dep_01_active value = 0 } remove_variable = bfl_dep_01_party }
}
}
bfl_pay_demand_interest_01 = {
