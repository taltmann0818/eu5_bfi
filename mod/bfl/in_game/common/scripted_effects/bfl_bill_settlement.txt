bfl_redeem_bill_01 = {
if = { limit = { var:bfl_con_01_active = 1 var:bfl_con_01_kind = 1 var:bfl_con_01_state = 1
    NOT = { has_variable = bfl_con_01_pledge }
    exists = var:bfl_con_01_party var:bfl_con_01_party = { bfl_financial_recipient = yes } var:bfl_con_01_party = scope:bfl_party
    bfl_period >= var:bfl_con_01_maturity_period gold >= var:bfl_con_01_basis
}
    save_scope_as = bfl_bank
    set_variable = { name = bfl_payment value = var:bfl_con_01_basis }
    add_gold = bfl_negative_payment
    scope:bfl_party = { add_gold = scope:bfl_bank.var:bfl_payment
        if = { limit = { bfl_has_accounts = yes } bfl_initialize_monetary = yes
            change_variable = { name = bfl_money_assets subtract = scope:bfl_bank.var:bfl_payment } } }
    change_variable = { name = bfl_other_payables subtract = var:bfl_payment }
    change_variable = { name = bfl_cash_committed subtract = var:bfl_payment }
    set_variable = { name = bfl_con_01_state value = 3 }
    set_variable = { name = bfl_con_01_active value = 0 }
    set_variable = { name = bfl_con_01_last_settlement value = bfl_period }
}
}
