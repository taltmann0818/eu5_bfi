bfl_offer_bond_workout_01 = {
if = { limit = { bfl_financial_recipient = yes bfl_bond_workout_researched = yes
            var:bfl_bnd_01_active ?= 1 exists = var:bfl_bnd_01_holder country_exists = var:bfl_bnd_01_holder
            NOT = { this = var:bfl_bnd_01_holder } }
            save_scope_as = bfl_bond_debtor
            if = { limit = { bfl_has_valid_controller = yes } overlord = { save_scope_as = bfl_bond_workout_authority } }
            else_if = { limit = { has_variable = bfl_resolution_estate } var:bfl_resolution_trustee = { save_scope_as = bfl_bond_workout_authority } }
            else = { save_scope_as = bfl_bond_workout_authority }
            var:bfl_bnd_01_holder = {
                if = { limit = { NOT = { has_variable = bfl_bond_workout_revision } }
                    set_variable = { name = bfl_bond_workout_revision value = 0 } }
                change_variable = { name = bfl_bond_workout_revision add = 1 }
                save_scope_value_as = { name = bfl_bond_workout_quote value = var:bfl_bond_workout_revision }
                set_variable = { name = bfl_bond_workout_issuer value = scope:bfl_bond_debtor }
                set_variable = { name = bfl_bond_workout_authority value = scope:bfl_bond_workout_authority }
                set_variable = { name = bfl_bond_workout_slot value = 1 }
                set_variable = { name = bfl_bond_workout_generation value = scope:bfl_bond_debtor.var:bfl_bnd_01_generation }
                set_variable = { name = bfl_bond_workout_face value = scope:bfl_bond_debtor.var:bfl_bnd_01_face }
                set_variable = { name = bfl_bond_workout_due value = scope:bfl_bond_debtor.var:bfl_bnd_01_due }
                set_variable = { name = bfl_bond_workout_rate value = scope:bfl_bond_debtor.var:bfl_bnd_01_rate }
                set_variable = { name = bfl_bond_workout_expiry value = bfl_period }
                set_variable = { name = bfl_bond_workout_secured value = 0 }
                trigger_event_non_silently = bfl_bond_workout.1
            } }
}
bfl_apply_bond_workout_01 = {
if = { limit = {
            var:bfl_bnd_01_active = 1 var:bfl_bnd_01_holder ?= scope:bfl_bond_creditor
            var:bfl_bnd_01_generation = scope:bfl_bond_creditor.var:bfl_bond_workout_generation
            var:bfl_bnd_01_face = scope:bfl_bond_creditor.var:bfl_bond_workout_face
            var:bfl_bnd_01_due = scope:bfl_bond_creditor.var:bfl_bond_workout_due
            var:bfl_bnd_01_rate = scope:bfl_bond_creditor.var:bfl_bond_workout_rate
            OR = { scope:bfl_bond_creditor.var:bfl_bond_workout_secured ?= 0
                AND = { scope:bfl_bond_creditor.var:bfl_bond_workout_secured ?= 1 bfl_bond_security_available = yes } } }
            set_variable = { name = bfl_bond_workout_loss value = { value = var:bfl_bnd_01_basis multiply = 0.20 } }
            set_variable = { name = bfl_bnd_01_face value = { value = var:bfl_bnd_01_face multiply = 0.80 multiply = 100 floor = yes divide = 100 } }
            set_variable = { name = bfl_bnd_01_basis value = { value = var:bfl_bnd_01_basis multiply = 0.80 } }
            set_variable = { name = bfl_bnd_01_due value = { value = var:bfl_bnd_01_due multiply = 0.80 multiply = 100 floor = yes divide = 100 } }
            set_variable = { name = bfl_bnd_01_rate value = { value = var:bfl_bnd_01_rate subtract = 0.01 min = 0.01 } }
            set_variable = { name = bfl_bnd_01_remainder value = 0 }
            change_variable = { name = bfl_bnd_01_generation add = 1 }
            if = { limit = { scope:bfl_bond_creditor.var:bfl_bond_workout_secured ?= 1 }
                set_variable = { name = bfl_revenue_bond_slot value = 1 }
                set_variable = { name = bfl_revenue_bond_expiry value = { value = bfl_period add = 24 } }
                set_variable = { name = bfl_revenue_concession_holder value = scope:bfl_bond_creditor }
                create_relation = { type = relation_type:bfl_workout_trade_access first = this second = scope:bfl_bond_creditor }
            }
            if = { limit = { var:bfl_bond_offer_slot = 1 } set_variable = { name = bfl_bond_offer_slot value = 0 } }
            scope:bfl_bond_creditor = {
                remove_variable = bfl_bond_workout_issuer
                if = { limit = { bfl_has_accounts = yes }
                    change_variable = { name = bfl_loss_carryforward add = scope:bfl_bond_debtor.var:bfl_bond_workout_loss }
                    bfl_refresh_asset_statement = yes bfl_evaluate_distress = yes }
            }
        }
}
