bfl_begin_resolution = {
bfl_initialize_resolution = yes
        if = { limit = { bfl_has_accounts = yes var:bfl_state < 4 }
            bfl_accrue_resolution_bonds = yes
            set_variable = { name = bfl_state value = 4 }
            set_variable = { name = bfl_resolution_started value = bfl_period }
            change_variable = { name = bfl_resolution_revision add = 1 }
            set_variable = { name = bfl_accept_deposits value = 0 }
            set_variable = { name = bfl_accept_equity value = 0 }
            set_variable = { name = bfl_equity_offer_active value = 0 }
            set_variable = { name = bfl_facility_mode value = 0 }
            set_variable = { name = bfl_goods_mandate value = 0 }
            bfl_leave_clearing = yes
        }
}
bfl_resolution_pay_01 = {
if = { limit = { var:bfl_dep_01_active = 1 var:bfl_resolution_pool > 0 var:bfl_resolution_budget > 0 }
    # Full funding pays the exact native fixed-point claim. Cent
    # rounding applies only to a partial, proportional recovery.
    # Otherwise sub-cent claims can keep a solvent bank in resolution forever.
    if = { limit = { var:bfl_resolution_budget >= var:bfl_resolution_pool }
        set_variable = { name = bfl_dep_01_recovery_held value = { value = var:bfl_dep_01_principal add = var:bfl_dep_01_interest_due } }
    }
    else = { change_variable = { name = bfl_dep_01_recovery_held add = { value = var:bfl_dep_01_principal add = var:bfl_dep_01_interest_due
        subtract = var:bfl_dep_01_recovery_held multiply = var:bfl_resolution_budget divide = var:bfl_resolution_pool
        multiply = 100 floor = yes divide = 100 } } } }
if = { limit = { var:bfl_dep_01_active = 1 exists = var:bfl_dep_01_party var:bfl_dep_01_party = { bfl_financial_recipient = yes }
    gold >= { value = bfl_resolution_other_reserves add = bfl_resolution_held } }
    var:bfl_dep_01_party = { set_variable = { name = bfl_pool value = scope:bfl_resolution_bank.var:bfl_dep_01_pool } }
    if = { limit = { var:bfl_dep_01_party = { bfl_valid_deposit_pool = yes } }
        set_variable = { name = bfl_payment value = { value = var:bfl_dep_01_recovery_held max = gold min = 0 } }
        add_gold = bfl_negative_payment
        var:bfl_dep_01_party = { bfl_credit_depositor = yes }
        set_variable = { name = bfl_interest_paid value = { value = var:bfl_payment max = var:bfl_dep_01_interest_due } }
        if = { limit = { var:bfl_dep_01_pool = 0 } var:bfl_dep_01_party = { bfl_record_withdrawn_interest = yes } }
        change_variable = { name = bfl_dep_01_interest_due subtract = var:bfl_interest_paid }
        change_variable = { name = bfl_dep_01_principal subtract = { value = var:bfl_payment subtract = var:bfl_interest_paid } }
        set_variable = { name = bfl_dep_01_withdrawal_requested value = { value = var:bfl_dep_01_principal add = var:bfl_dep_01_interest_due } }
        change_variable = { name = bfl_resolution_paid add = var:bfl_payment }
        change_variable = { name = bfl_dep_01_recovery_held subtract = var:bfl_payment }
        change_variable = { name = bfl_dep_01_generation add = 1 }
        if = { limit = { var:bfl_dep_01_principal <= 0 var:bfl_dep_01_interest_due <= 0 }
            set_variable = { name = bfl_dep_01_active value = 0 } }
        var:bfl_dep_01_party = { if = { limit = { bfl_has_accounts = yes }
            bfl_refresh_asset_statement = yes bfl_evaluate_distress = yes } }
        save_scope_as = bfl_bank
    }
}

bfl_authorize_resolution_guarantee = {
bfl_initialize_resolution = yes
        if = { limit = { bfl_resolution_guarantee_authority = yes var:bfl_state < 4
            NOT = { has_variable = bfl_resolution_guarantor } overlord = { gold >= 100 } }
            set_variable = { name = bfl_resolution_guarantor value = overlord }
            set_variable = { name = bfl_resolution_guarantee_left value = 100 }
            overlord = { if = { limit = { NOT = { has_variable = bfl_guarantee_exposure } }
                set_variable = { name = bfl_guarantee_exposure value = 0 } }
                change_variable = { name = bfl_guarantee_exposure add = 100 } }
        }
}
bfl_pay_resolution_guarantee = {
if = { limit = { var:bfl_resolution_guarantee_due > 0 exists = var:bfl_resolution_guarantor
    country_exists = var:bfl_resolution_guarantor
    exists = var:bfl_resolution_guarantee_beneficiary var:bfl_resolution_guarantee_beneficiary = { bfl_financial_recipient = yes } }
    save_scope_as = bfl_resolution_bank
    var:bfl_resolution_guarantee_beneficiary = {
        set_variable = { name = bfl_pool value = scope:bfl_resolution_bank.var:bfl_resolution_guarantee_pool } }
    if = { limit = { var:bfl_resolution_guarantee_beneficiary = { bfl_valid_deposit_pool = yes }
        var:bfl_resolution_guarantor.gold >= var:bfl_resolution_guarantee_due }
        set_variable = { name = bfl_payment value = var:bfl_resolution_guarantee_due }
        var:bfl_resolution_guarantor = { add_gold = { value = scope:bfl_resolution_bank.var:bfl_payment multiply = -1 }
            change_variable = { name = bfl_guarantee_exposure subtract = scope:bfl_resolution_bank.var:bfl_payment } }
        save_scope_as = bfl_bank
        var:bfl_resolution_guarantee_beneficiary = { bfl_credit_depositor = yes
            if = { limit = { bfl_has_accounts = yes }
                change_variable = { name = bfl_profit_credit add = scope:bfl_resolution_bank.var:bfl_payment } } }
        change_variable = { name = bfl_resolution_guarantee_paid add = var:bfl_payment }
        change_variable = { name = bfl_resolution_guarantee_left subtract = var:bfl_payment }
        set_variable = { name = bfl_resolution_guarantee_due value = 0 }
    }
}
