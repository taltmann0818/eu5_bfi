    # Root bank. Only its live landed controller can be the primary issuer.
    bfl_underwrite_sovereign_bond = {
        if = { limit = { bfl_can_originate_bond_underwriting = yes bfl_has_valid_controller = yes
            bfl_branch_access = { PARTY = overlord }
            NOT = { has_variable = bfl_bond_guard }
            var:bfl_bond_amount > 0
            bfl_free_cash >= var:bfl_bond_amount bfl_capital_headroom >= var:bfl_bond_amount
        }
            save_scope_as = bfl_bond_underwriter
            overlord = { bfl_initialize_bonds = yes save_scope_as = bfl_bond_issuer }
            if = { limit = { scope:bfl_bond_issuer = {
                country_type = location monthly_income_total > 0 bfl_has_bond_slot = yes bfl_bond_arrears = 0
                monthly_income_total >= { value = bfl_bond_monthly_coupon add = { value = scope:bfl_bond_underwriter.var:bfl_bond_amount multiply = var:bfl_bond_rate divide = 12 } multiply = 4 }
                bfl_bond_capacity >= scope:bfl_bond_underwriter.var:bfl_bond_amount
                bfl_bond_remaining_monthly >= scope:bfl_bond_underwriter.var:bfl_bond_amount NOT = { has_variable = bfl_bond_guard }
                var:bfl_bond_rate >= 0.01 var:bfl_bond_rate <= 0.06
            } }
                set_variable = bfl_bond_guard
                scope:bfl_bond_issuer = {
                    set_variable = bfl_bond_guard
                    bfl_allocate_sovereign_bond = yes
                    add_gold = scope:bfl_bond_underwriter.var:bfl_bond_amount
                    set_variable = { name = bfl_bond_last_issue value = bfl_period }
                    if = { limit = { var:bfl_bond_financing_period != bfl_period }
                        set_variable = { name = bfl_bond_financing_period value = bfl_period }
                        set_variable = { name = bfl_bond_financing value = 0 } }
                    change_variable = { name = bfl_bond_financing add = scope:bfl_bond_underwriter.var:bfl_bond_amount }
                    remove_variable = bfl_bond_guard
                }
                add_gold = { value = var:bfl_bond_amount multiply = -1 }
                set_variable = { name = bfl_facility_paid value = var:bfl_bond_amount }
                set_variable = { name = bfl_facility_blocked value = 0 }
                remove_variable = bfl_bond_guard
                bfl_refresh_asset_statement = yes
            }
        }
    }
    bfl_run_controller_bond_facility = {
        set_variable = { name = bfl_facility_paid value = 0 }
        set_variable = { name = bfl_facility_blocked value = 1 }
        if = { limit = { bfl_has_valid_controller = yes var:bfl_facility_mode > 0
            OR = { NOT = { has_variable = bfl_facility_last_period } bfl_period > var:bfl_facility_last_period } }
            set_variable = { name = bfl_facility_last_period value = bfl_period }
            save_scope_as = bfl_bond_underwriter
            overlord = {
                # Native monthly_balance is LAST realized balance. Sample the
                # live forecast once here, never in a mass-evaluated AI weight.
                set_variable = { name = bfl_fiscal_forecast value = tutorial_balance }
                set_variable = { name = bfl_bond_target value = { value = monthly_income_total subtract = var:bfl_fiscal_forecast min = 0 multiply = 0.6 } }
                if = { limit = { scope:bfl_bond_underwriter.var:bfl_facility_mode = 2 }
                    set_variable = { name = bfl_bond_target value = { value = 200 subtract = gold subtract = var:bfl_fiscal_forecast min = 0 } } }
            }
            # Size down to live underwriting capacity instead of rejecting a
            # whole recurring request when only a smaller draw is affordable.
            set_variable = { name = bfl_bond_amount value = { value = overlord.var:bfl_bond_target max = overlord.bfl_bond_draw_max multiply = 100 floor = yes divide = 100 } }
            set_variable = { name = bfl_facility_requested value = var:bfl_bond_amount }
            bfl_underwrite_sovereign_bond = yes
            if = { limit = { var:bfl_facility_paid > 0 }
                if = { limit = { NOT = { has_variable = bfl_facility_draws } } set_variable = { name = bfl_facility_draws value = 0 } }
                change_variable = { name = bfl_facility_draws add = 1 } }
        }
        set_variable = { name = bfl_facility_reason value = 0 }
        if = { limit = { var:bfl_facility_requested > var:bfl_facility_paid }
            set_variable = { name = bfl_facility_reason value = 1 }
            if = { limit = { bfl_free_cash < var:bfl_facility_requested } set_variable = { name = bfl_facility_reason value = 2 } }
            else_if = { limit = { bfl_capital_headroom < var:bfl_facility_requested } set_variable = { name = bfl_facility_reason value = 3 } }
            else_if = { limit = { overlord.bfl_bond_capacity < var:bfl_facility_requested } set_variable = { name = bfl_facility_reason value = 4 } }
            else_if = { limit = { overlord.bfl_bond_arrears > 0 } set_variable = { name = bfl_facility_reason value = 5 } }
            else_if = { limit = { overlord.var:bfl_bond_last_issue = bfl_period } set_variable = { name = bfl_facility_reason value = 6 } }
            else_if = { limit = { overlord = { NOT = { bfl_has_bond_slot = yes } } } set_variable = { name = bfl_facility_reason value = 7 } }
            if = { limit = { NOT = { var:bfl_facility_last_reason ?= var:bfl_facility_reason } }
                set_variable = { name = bfl_feedback_shortfall value = { value = var:bfl_facility_requested subtract = var:bfl_facility_paid } }
                bfl_record_feedback = { KIND = 9 AMOUNT = var:bfl_feedback_shortfall } }
        }
        set_variable = { name = bfl_facility_last_reason value = var:bfl_facility_reason }
    }
