request_loan = { 
	type = diplomacy
	category = CATEGORY_ECONOMY_ACTIONS

	ai_tick = never
	
	potential = {
	}

	allow = {
	}

	diplo_chance = {
		base = 5
		rank = 0
		opinion = 0.2
		relative_strength = 0
		border_distance = 0
		trust_in_actor = 0.2
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		column = {
			data = country_diplomacy_info_loan
		}
		visible = {
			exists = capital
			capital = { is_discovered_by = scope:actor }
			within_diplomatic_range = scope:actor
			has_reform = government_reform:banking_country
		}
        enabled = {
            bfl_branch_access = { PARTY = scope:actor } bfl_native_request_funded = yes
			gold > monthly_income_total
		}
	}

	effect = {
	
		scope:recipient ?= {
        if = { limit = { bfl_branch_access = { PARTY = scope:actor } bfl_native_request_funded = yes }
	
			give_loan = {
				target = scope:actor
				amount = {
					value = scope:recipient.monthly_income_total
					max = {
						value = scope:actor.monthly_income_trade_and_tax
						multiply = 2
					}
				}
				interest = scope:actor.modifier:bank_interest
				months = 60
			}
		
 if = { limit = { bfl_has_accounts = yes } bfl_reconcile_native_assets = yes }

        }
}
	}
	ai_will_do = -10
}
