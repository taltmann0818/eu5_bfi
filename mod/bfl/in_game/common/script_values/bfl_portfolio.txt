bfl_treasury_portfolio_units = {
value = 0 if = { limit = { country_type = location bfl_has_management = yes
        exists = var:bfl_bank var:bfl_bank = { has_variable = bfl_ownership_schema
            var:bfl_state_owner ?= prev var:bfl_state_units > 0 } } value = var:bfl_bank.var:bfl_state_units }
}
bfl_treasury_portfolio_market = {
value = 0 if = { limit = { country_type = location bfl_has_management = yes
        exists = var:bfl_bank var:bfl_bank = { has_variable = bfl_ownership_schema
            var:bfl_state_owner ?= prev var:bfl_state_units > 0 } } value = var:bfl_bank.bfl_equity_quote multiply = var:bfl_bank.var:bfl_state_units }
}
bfl_portfolio_market = {
value = bfl_exchange_market add = bfl_treasury_portfolio_market
}
bfl_treasury_portfolio_basis = {
value = 0 if = { limit = { country_type = location bfl_has_management = yes
        exists = var:bfl_bank var:bfl_bank = { has_variable = bfl_ownership_schema
            var:bfl_state_owner ?= prev var:bfl_state_units > 0 } } value = var:bfl_bank.var:bfl_state_basis }
}
bfl_portfolio_basis = {
value = bfl_exchange_basis add = bfl_treasury_portfolio_basis
}
bfl_portfolio_return = {
value = 0 if = { limit = { bfl_portfolio_basis > 0 }
        value = bfl_portfolio_market divide = bfl_portfolio_basis subtract = 1 }
}
bfl_treasury_portfolio_return = {
value = 0 if = { limit = { bfl_treasury_portfolio_basis > 0 }
        value = bfl_treasury_portfolio_market divide = bfl_treasury_portfolio_basis subtract = 1 }
}
bfl_treasury_portfolio_weight = {
value = 0 if = { limit = { bfl_portfolio_market > 0 }
        value = bfl_treasury_portfolio_market divide = bfl_portfolio_market }
}
bfl_portfolio_realized = {
value = bfl_exchange_realized
        if = { limit = { has_variable = bfl_treasury_equity_realized } add = var:bfl_treasury_equity_realized }
}
